=================================================
ANSIBLE PLAYBOOK
=================================================

---
- name: Install tools based on OS
  hosts: all
  become: yes

  tasks:

    - name: Install htop on RedHat
      ansible.builtin.yum:
        name: htop
        state: present
      when: ansible_facts.os_family == "RedHat"


    - name: Install htop on Debian
      ansible.builtin.apt:
        name: htop
        state: present
        update_cache: yes
      when: ansible_facts.os_family == "Debian"



=================================================
TERRAFORM ROUTE TABLE
=================================================

resource "aws_route_table" "main_rt" {

  vpc_id = aws_vpc.main.id

  tags = {
    Name = "main-route-table"
  }
}


resource "aws_route" "internet" {

  route_table_id = aws_route_table.main_rt.id

  destination_cidr_block = "0.0.0.0/0"

  gateway_id = aws_internet_gateway.main.id
}


resource "aws_route_table_association" "assoc" {

  subnet_id = aws_subnet.public.id

  route_table_id = aws_route_table.main_rt.id
}

=================================================
JENKINS PIPELINE
=================================================

pipeline {

 agent any


 stages {


  stage('Checkout') {

    steps {

      git 'https://github.com/Pathnex/sample-java-app.git'

    }
  }


  stage('Build') {

    steps {

      sh '''
      mvn -Dmaven.repo.local=.m2/repository clean package
      '''

    }
  }


  stage('Test') {

    steps {

      sh 'mvn test'

    }
  }

 }

}
